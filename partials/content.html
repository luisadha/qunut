reset="\033[0m" # Cek ketersediaan binary pager if command -v less > /dev/null 2>&1; then printf "${blue}[info]: ${yellow}less is installed and likely used as the default pager.${reset}\n" elif command -v more > /dev/null 2>&1; then printf "${blue}[info]: ${yellow}more is installed and likely used as the default pager.${reset}\n" else printf "${red}No common pager (less/more) found.${reset}\n" fi # Cek variabel lingkungan MANPAGER if [ -n "$MANPAGER" ];
